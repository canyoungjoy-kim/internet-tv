rules_version = '2';
service cloud.firestore {
  match /databases/{db}/documents {

    // 관리자 = 구글 로그인 사용자, 입주민 = 익명 로그인 사용자
    // 관리자 = 소장님 Google 계정 (아래 주소를 바꾸면 관리자가 바뀝니다)
    function isAdmin() {
      return request.auth != null
          && request.auth.token.email == 'canyoungjoy@gmail.com'
          && request.auth.token.firebase.sign_in_provider != 'anonymous';
    }

    // 직원·기사 공용 계정
    function isStaff() { return request.auth != null && request.auth.token.email == 'staff@thetime.kr'; }
    function isTech()  { return request.auth != null && request.auth.token.email == 'tech@thetime.kr'; }
    function isWorker(){ return isStaff() || isTech(); }

    // 작업자가 고칠 수 있는 항목만 나열 — 이 외의 필드를 건드리면 서버가 거부
    function changed() {
      return request.resource.data.diff(resource.data).affectedKeys();
    }
    // 설치기사 — 작업 결과만 입력. 일정·개인정보는 손댈 수 없습니다
    function onlyWorkFields() {
      return changed().hasOnly(['ret','ins','sn','status','note','updatedAt','pin'])
             // 비밀번호는 '지우는 것'만 허용 (작업완료 시 자동 삭제)
             && (!changed().hasAny(['pin']) || request.resource.data.pin == '');
    }
    // 관리실 직원 — 위에 더해 '입주민 예약을 세대에 반영'하는 데 필요한 일정 항목까지
    // (이름·연락처는 units 문서에 아예 없으므로 여기서도 볼 수 없습니다)
    function staffFields() {
      return changed().hasOnly(['ret','ins','sn','status','note','updatedAt','pin',
                                'date','slot','tech','presence',
                                'bookedBy','bookedAt','bookedAtMs']);
    }

    match /sites/{site} {

      // 작업 자료 — 이름·연락처는 여기에 없습니다
      match /units/{unitId} {
        allow read:   if isAdmin() || isWorker();
        allow create, delete: if isAdmin();
        allow update: if isAdmin()
                      || (isStaff() && staffFields())
                      || (isTech()  && onlyWorkFields());
      }

      // 입주민 이름·연락처 — 관리자와 직원만. 기사는 읽기 자체가 불가
      match /contacts/{unitId} {
        allow read:  if isAdmin() || isStaff();
        allow write: if isAdmin();
      }

      // 작업일·시간대·기사 명단 — 작업자는 보기만
      match /meta/config {
        allow read:  if isAdmin() || isWorker();
        allow write: if isAdmin();
      }
      // 예약 페이지가 읽는 공개 정보 — 개인정보 없음(작업일·시간대·정원)
      //  ※ 관리자만 씁니다. 작업일·시간대는 관리자만 바꿀 수 있는데, 옛 일정이 남은
      //    직원 기기가 덮어쓰면 입주민 화면에 없앤 날짜가 되살아납니다.
      match /meta/public {
        allow read: if request.auth != null;
        allow write: if isAdmin();
      }
      // 본인확인용 해시 목록 — 원본 번호는 들어있지 않음
      //  ※ 관리자만 씁니다. 연락처를 관리하는 계정이 관리자뿐인데 직원 기기가
      //    이 문서를 덮어쓰면 명단이 빈 채로 올라가 본인확인이 통째로 풀립니다.
      match /meta/roster {
        allow read: if request.auth != null;
        allow write: if isAdmin();
      }
      // 시간대별 예약 인원 (숫자만)
      match /meta/avail {
        allow read: if request.auth != null;
        allow write: if isAdmin() || isStaff();
      }

      // 입주민 예약 접수함
      //  문서 ID = 호실_연락처뒷4자리 → 뒷4자리를 모르면 남의 예약을 열 수 없습니다
      //  목록 조회(list)는 관리자만 → 통째로 긁어갈 수 없습니다
      match /bookings/{bookingId} {
        allow get: if request.auth != null;
        // 접수가 열려 있을 때만, 그리고 문서ID가 호실_뒷4자리와 일치할 때만
        allow create, update: if request.auth != null
          && bookingId == request.resource.data.ho + '_' + request.resource.data.phone4
          && (isAdmin() || isStaff()
              || get(/databases/$(db)/documents/sites/$(site)/meta/public).data.bookingOpen == true);
        // 관리사무소(관리자·직원)는 예약함 전체를 받아 세대에 반영해야 합니다.
        // 설치기사는 목록 조회 불가 — 세대별 작업요청서에 필요한 정보만 받습니다.
        allow list:   if isAdmin() || isStaff();
        allow delete: if isAdmin();
      }

      // 시간대별 예약 인원 — 정원 초과를 서버에서 막기 위한 카운터
      //  입주민이 트랜잭션으로 직접 올리고 내립니다(개인정보 없음)
      //  ※ 예전에는 로그인만 하면 아무 숫자나 쓸 수 있어, 9999 를 넣으면
      //    전 시간대가 '마감'으로 보이게 만들 수 있었습니다.
      //    이제 입주민은 자기 예약 한 건만큼(±1)만 움직일 수 있습니다.
      match /slots/{slotId} {
        allow read: if request.auth != null;
        allow write: if isAdmin() || isStaff()
          || (request.auth != null
              && request.resource.data.keys().hasOnly(['date','slot','cap','n','at'])
              && request.resource.data.n is int
              && request.resource.data.n >= 0
              && (resource == null
                    ? request.resource.data.n <= 1
                    : (request.resource.data.n >= resource.data.n - 1
                       && request.resource.data.n <= resource.data.n + 1
                       /* 정원(cap)은 입주민이 건드릴 수 없습니다.
                          예약수(n)만 ±1 로 잡아두고 정원을 열어두면,
                          정원을 999 로 올려 한 시간대에 무제한으로 몰리게 하거나
                          0 으로 내려 멀쩡한 시간대를 마감시킬 수 있습니다.
                          관리자·직원은 위쪽 조건으로 통과하므로 영향이 없습니다. */
                       && request.resource.data.cap == resource.data.cap)));
      }
    }
  }
}
